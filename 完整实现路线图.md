### M2: TypeChecker双向推导 (2-3小时)

**核心机制**:

```cpp
class TypeChecker {
private:
    Type* expected_type_ = nullptr;  // 期望类型（从上下文）
    
public:
    void visit(VarDecl* node) {
        // 设置期望类型
        if (node->getType()) {
            expected_type_ = node->getType();
        }
        
        // 检查初始值
        node->getInit()->accept(this);
        
        expected_type_ = nullptr;  // 清空
    }
    
    void visit(ClosureExpr* node) {
        // 如果有期望类型且是函数类型
        if (expected_type_ && expected_type_->isFunction()) {
            auto* fn_type = static_cast<FunctionType*>(expected_type_);
            
            // 推导无类型参数
            for (size_t i = 0; i < node->getParams().size(); i++) {
                auto& param = node->getParams()[i];
                if (!param.type && i < fn_type->getParamTypes().size()) {
                    param.type = fn_type->getParamTypes()[i];  // 推导！
                }
            }
        }
        
        // 检查所有参数都有类型
        for (const auto& param : node->getParams()) {
            if (!param.type) {
                error("Cannot infer parameter type");
            }
        }
        
        // ... 继续类型检查
    }
};
```

### M4: 泛型实例化算法 (2-3小时)

**核心算法**:
```cpp
Type* TypeSystem::instantiateGeneric(
    const std::string& template_name,
    const std::vector<Type*>& type_args
) {
    // 1. 生成实例名: "Option" + ["i32"] -> "Option_i32"
    std::string instance_name = generateInstanceName(template_name, type_args);
    
    // 2. 查找缓存
    if (auto it = instantiated_types_.find(instance_name)) {
        return it->second;
    }
    
    // 3. 查找模板
    auto* tmpl = lookupGenericTemplate(template_name);
    if (!tmpl) return nullptr;
    
    // 4. 替换类型参数
    // 复制definition，替换所有T为i32
    auto* instantiated_def = cloneAndSubstitute(tmpl->definition, tmpl->type_params, type_args);
    
    // 5. 创建具体类型
    Type* instance_type = createTypeFromDefinition(instance_name, instantiated_def);
    
    // 6. 注册
    instantiated_types_[instance_name] = instance_type;
    registerEnum(instance_type);
    
    return instance_type;
}
```

### M5: Parser泛型使用解析 (1小时)

**修改parseType()**:
```cpp
Type* Parser::parseType() {
    Token base = consume(IDENTIFIER);
    std::string base_name = base.lexeme;
    
    // 检查是否有泛型参数 <...>
    if (match(TokenType::LESS)) {
        std::vector<Type*> type_args;
        do {
            type_args.push_back(parseType());
        } while (match(TokenType::COMMA));
        consume(TokenType::GREATER);
        
        // 请求泛型实例化
        return type_system_->instantiateGeneric(base_name, type_args);
    }
    
    // 普通类型查找
    return type_system_->lookupType(base_name);
}
```
