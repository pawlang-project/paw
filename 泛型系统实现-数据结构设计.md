## 🎯 目标

**实现前**:
```paw
type Option<T> = enum { Some(T), None };
// ❌ 不支持泛型定义
```

**实现后**:
```paw
type Option<T> = enum { Some(T), None };
// ✅ 注册为泛型模板

let opt: Option<i32> = Option::Some(42);
// ✅ 实例化为Option_i32
```

## 📐 数据结构设计

### GenericTemplate

```cpp
// src/middleend/types/generic_template.h（新文件）

struct GenericTemplate {
    std::string name;                      // "Option"
    std::vector<std::string> type_params;  // ["T"]
    
    // 保存原始定义（用于实例化）
    enum DefinitionKind {
        ENUM,
        STRUCT,
        // FUNCTION  // 未来支持
    } kind;
    
    union {
        EnumDecl* enum_def;
        StructDecl* struct_def;
    };
    
    GenericTemplate(std::string n, std::vector<std::string> params, EnumDecl* def)
        : name(std::move(n)), type_params(std::move(params)), 
          kind(ENUM), enum_def(def) {}
};
```

### TypeSystem扩展

```cpp
class TypeSystem {
public:
    // 泛型模板管理
    void registerGenericTemplate(GenericTemplate* tmpl);
    GenericTemplate* lookupGenericTemplate(const std::string& name);
    
    // 泛型实例化
    Type* instantiateGeneric(const std::string& name, 
                            const std::vector<Type*>& type_args);
    
private:
    // 泛型模板表
    std::unordered_map<std::string, GenericTemplate*> generic_templates_;
    
    // 已实例化类型缓存: "Option_i32" -> Type*
    std::unordered_map<std::string, Type*> instantiated_types_;
};
```
