### 模式匹配 ⭐ v0.2.2 增强

```paw
// 基础匹配
x is {
    Some(v) => v,
    None => 0,
    _ => default,
}

// 范围模式 (Range Pattern)
score is {
    90..100 => "A",      // 90 <= score < 100
    80..=90 => "B",      // 80 <= score <= 90 (包含式)
    'a'..'z' => "小写",   // 字符范围
    _ => "其他"
}

// OR 模式 (Or Pattern)
x is {
    1 | 2 | 3 => "小",
    4 | 5 | 6 => "中",
    _ => "大"
}

// 中间 Rest 模式 (Middle Rest)
arr is {
    [first, .., last] => first + last,     // 匹配首尾
    [a, b, .., y, z] => a + b + y + z,     // 前缀+后缀
    [.., last] => last,                     // 仅尾部
    _ => 0
}

// 嵌套模式 (Nested Pattern)
tuple is {
    ((a, b), (c, d)) => a + b + c + d,     // 嵌套元组
    ([x, y], [z, w]) => x + y + z + w,     // 元组中的数组
    _ => 0
}

// 守卫模式 (Guard Pattern)
x is {
    n if n > 10 => "大于10",                // 条件守卫
    n if n > 0 => "正数",
    0 => "零",
    _ => "负数"
}

// 组合使用
arr is {
    [a, b, c] if a + b + c > 50 => 100,    // 守卫 + 数组
    [first, ..] if first > 10 => 200,      // 守卫 + Rest
    _ => 300
}
```

### Optional和Result ⭐ v1.4.2

```paw
// Optional (T?) - 可选值
let name: string? = some("Alice");  // some() 构造
let empty: i32? = none;             // none 空值
if empty == none { }                // none 检查

// 模式匹配
name is {
    some(val) => val,
    none => "Unknown"
}

// Result (T!) - 错误处理
fn divide(a: i32, b: i32) -> i32! {
    if b == 0 { return err("error"); }
    return ok(a / b);
}
let x = divide(10, 2)!;  // Try表达式

// 组合类型
fn db_query() -> User?! {  // T?! = Result<Optional<T>>
    return ok(none);  // 或 ok(null)
}
```
